# Q1

Staff_Evaluations = []

for i in range(6):
    nama = input(f"Nama pekerja {i+1}: ")
    m = []

    for j in ("Prestasi", "Kehadiran"):
        x = float(input(f"Markah {j} (0-100): "))

        while not 0 <= x <= 100:
            x = float(input("Salah! Masukkan semula (0-100): "))

        m.append(x)

    akhir = m[0] * 0.6 + m[1] * 0.4

    if akhir >= 80:
        s = "Cemerlang (Elaun RM 500)"
    elif akhir >= 60:
        s = "Baik (Elaun RM 250)"
    else:
        s = "Perlu Penambahbaikan (Tiada Elaun)"

    Staff_Evaluations.append([nama, akhir, s])

for n, a, s in Staff_Evaluations:
    print(f"{n}: {a:.1f} - {s}")

print(f"Purata: {sum(e[1] for e in Staff_Evaluations) / 6:.2f}")


# Q2

HARGA = {"Student": 80, "Profesional": 200, "VIP": 450}


def kira_yuran_bersih(kategori_counts, total_peserta):
    kasar = sum(HARGA[k] * v for k, v in kategori_counts.items())
    return kasar, (kasar * 0.9 if total_peserta >= 3 else kasar)


def main():
    counts = dict.fromkeys(HARGA, 0)
    n = int(input("Jumlah peserta: "))

    for _ in range(n):
        k = input("Kategori (Student/Profesional/VIP): ")
        while k not in HARGA:
            k = input("Tak sah! Student/Profesional/VIP: ")
        counts[k] += 1

    kasar, bersih = kira_yuran_bersih(counts, n)
    print(counts)
    print(f"Kasar: RM{kasar:.2f}")
    print(f"Diskaun: RM{kasar - bersih:.2f}")
    print(f"Bersih: RM{bersih:.2f}")


main()

import streamlit as st

jumlah = st.number_input("Jumlah Pinjaman (RM)", value=0.0)
kadar = st.number_input("Kadar Faedah Tahunan (%)", value=0.0)
tempoh = st.number_input("Tempoh Pinjaman (Tahun)", value=0.0)
jenis = st.selectbox(
    "Jenis Pinjaman",
    ["Kenderaan Baharu", "Kenderaan Terpakai", "Pinjaman Peribadi"],
)

if st.button("Kira Ansuran"):
    try:
        if jumlah <= 0 or kadar <= 0:
            raise ValueError("Jumlah pinjaman dan kadar faedah mesti > 0")
        faedah = jumlah * (kadar / 100) * tempoh
        ansuran = (jumlah + faedah) / (tempoh * 12)
    except ZeroDivisionError:
        st.error("Tempoh pinjaman tidak boleh 0!")
    except ValueError as e:
        st.warning(str(e))
    else:
        st.success(f"Ansuran Bulanan: RM{ansuran:.2f}")
        st.success(f"Jumlah Keseluruhan Bayaran: RM{jumlah + faedah:.2f}")

import numpy as np
import pandas as pd

data = {
    'Order_ID': [101, 102, 103, 101, 104, 105, 106, 107],
    'Category': ['Electronics', 'Fashion', 'Electronics', 'Electronics',
                 'Home', 'Fashion', 'Home', 'Electronics'],
    'Quantity': [2, 5, None, 2, 1, None, 4, 3],
    'Price_Per_Unit': [1500, 80, 200, 1500, 350, 80, 120, None],
    'Discount_Rate': [0.10, 0.05, 0.00, 0.10, 0.00, 0.05, 0.15, 0.10],
}
df_orders = pd.DataFrame(data)

df_orders = df_orders.drop_duplicates()
df_orders['Quantity'] = df_orders['Quantity'].fillna(
    df_orders['Quantity'].median()
)
df_orders['Price_Per_Unit'] = df_orders['Price_Per_Unit'].fillna(
    df_orders['Price_Per_Unit'].mean()
)
df_orders['Gross_Amount'] = df_orders['Quantity'] * df_orders['Price_Per_Unit']
df_orders['Net_Amount'] = df_orders['Gross_Amount'] * (1 - df_orders['Discount_Rate'])

hasil = df_orders[(df_orders['Quantity'] >= 2) & (df_orders['Net_Amount'] > 300)]
print(hasil.sort_values('Net_Amount', ascending=False))
print(df_orders.groupby('Category').agg({'Quantity': 'sum', 'Net_Amount': 'mean'}))
